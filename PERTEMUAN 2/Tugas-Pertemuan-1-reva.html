# --- Menentukan Nilai m ---
nilai_m <- c(3, 4, 5, 7, 10, 20)

  m = nilai_m,
  SSE = NA,
  MSE = NA,
  RMSE = NA,
  MAPE = NA
)

for (i in 1:length(nilai_m)) {

  m <- nilai_m[i]

  # Menghitung SMA pada data latih
  data.sma <- SMA(training$harga_close, n = m)

  # Mengambil nilai SMA terakhir
  nilai_terakhir <- tail(na.omit(data.sma), 1)

  # Forecast untuk seluruh data uji
  ramalan <- rep(nilai_terakhir, n_test)

  # Error
  error <- testing$harga_close - ramalan

  # SSE
  SSE <- sum(error^2)

  # MSE
  MSE <- mean(error^2)

  # RMSE
  RMSE <- sqrt(MSE)

  # MAPE
  MAPE <- mean(abs(error / testing$harga_close)) * 100

  # Masukkan hasil
  hasil_sma[i, "SSE"]  <- SSE
  hasil_sma[i, "MSE"]  <- MSE
  hasil_sma[i, "RMSE"] <- RMSE
  hasil_sma[i, "MAPE"] <- MAPE
}
